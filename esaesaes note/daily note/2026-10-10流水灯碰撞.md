```.c
void main()
{
	unsigned char ledl,ledr,temp;
	P2=0x7e;
	ledl=0x7f;//ledl=P2|0x0f;
	ledr=0xfe;//ledr=P2|0xf0
	while(1)
	{
		Delay(100);
	  if(P2==0xe7)
	  {
			temp = ledl;
      ledl = ledr;
			ledr = temp;
	  }
		ledl=(ledl>>1)|0x80;
		ledr=(ledr<<1)|0x01;
		P2=(ledl&ledr);
		if(P2==0x7e)
		{
			temp = ledl;
      ledl = ledr;
      ledr = temp;
		}
	}
}
```

- ledl和ledr单纯用来记录状态，位运算&取0低电平后赋值给P2;
- ledl和ledr初始值也可通过led|取1来得，
- 0111 1110 |0000 1111----->左边0111 1111
- 0111 1110|1111 0000------>右边1111 1110
